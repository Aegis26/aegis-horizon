# Add Product-Based Commission Tiers

**Goal:** Allow different commission percentages per employee based on product classification.

**Current:** Employee has 1 commission % (e.g., 10%)
**Needed:** Employee has multiple commission % per product type (e.g., 10% for Product A, 15% for Product B)

**Structure:**
```
Employee: John
  - Product Type 1 (Consulting): 10%
  - Product Type 2 (Software): 15%
  - Product Type 3 (Support): 8%

Employee: Sarah
  - Product Type 1 (Consulting): 12%
  - Product Type 2 (Software): 18%
  - Product Type 3 (Support): 10%
```

```typescript
import { useState, useEffect } from 'react'

export default function Settings() {
  const router = useRouter()
  const { orgId } = router.query

  const [teamMembers, setTeamMembers] = useState([])

  const [commissions, setCommissions] = useState({}) // { "userId-productTypeId": percentage }
  const [saving, setSaving] = useState(false)
  const [saveMessage, setSaveMessage] = useState('')

  // Fetch data
  useEffect(() => {
    const fetchData = async () => {
      if (!orgId) return

      try {
        // Get team members
        const membersResponse = await fetch(`/api/organizations/${orgId}/members`)
        const membersData = await membersResponse.json()
        setTeamMembers(membersData)

        // Get product types
        const productResponse = await fetch(`/api/product-types?orgId=${orgId}`)
        const productData = await productResponse.json()
        setProductTypes(productData)

        // Get commission settings
        const commissionsResponse = await fetch(`/api/commissions/settings?orgId=${orgId}`)
        const commissionsData = await commissionsResponse.json()
        
        // Convert to { "userId-productTypeId": percentage } format
        const commissionMap = {}
        commissionsData.forEach(c => {
          commissionMap[`${c.user_id}-${c.product_type_id}`] = c.commission_percentage
        })
        setCommissions(commissionMap)
      } catch (error) {
        console.error('Error fetching data:', error)
      }
    }

    fetchData()
  }, [orgId])

  // Handle commission % change
  const handleCommissionChange = (userId, productTypeId, value) => {
    const key = `${userId}-${productTypeId}`
    setCommissions(prev => ({
      ...prev,
      [key]: parseFloat(value) || 0
    }))
  }

  // Save commission settings
  const handleSaveCommissions = async () => {
    setSaving(true)
    setSaveMessage('')

    try {
      const response = await fetch(`/api/commissions/settings`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          orgId,
          commissions, // { "userId-productTypeId": percentage }
        }),
      })

      if (!response.ok) {
        const error = await response.json()
        setSaveMessage(`Error: ${error.error}`)
        setSaving(false)
        return

      setSaveMessage('Commission settings saved! ✅')
      setTimeout(() => setSaveMessage(''), 3000)
    } catch (error) {
      console.error('Error saving:', error)
      setSaveMessage('Failed to save commission settings')
    } finally {
      setSaving(false)
    }
  }

  // Only admins can see this
  if (userRole && userRole !== 'admin' && userRole !== 'owner') {
    return null
  }

  return (
    <div className="space-y-8">
      {/* ... existing sections ... */}

                {pt.name}
              </span>
            ))}
          </div>
        </div>

        {/* Commission Grid */}
        <div className="space-y-6">
```

```typescript
// When moving to "Closed Won", calculate commission based on PRODUCT TYPE
if (status === 'Closed Won' && opportunity.status !== 'Closed Won') {
  // Get the product type of this opportunity
  const productType = opportunity.product_type // e.g., 'consulting'
  
  // Find product type ID
  const productTypeRecord = await db.query.productTypes.findFirst({
    where: and(
      eq(productTypes.org_id, orgId),
      eq(productTypes.name, productType)
    ),
  })

  if (productTypeRecord) {
    // Get employee's commission % for THIS PRODUCT TYPE
    const commissionSetting = await db.query.employeeCommissions.findFirst({
      where: and(
        eq(employeeCommissions.org_id, orgId),
        eq(employeeCommissions.user_id, opportunity.owner_user_id),
        eq(employeeCommissions.product_type_id, productTypeRecord.id)
      ),
    })

    if (commissionSetting) {
      const commissionPercentage = parseFloat(commissionSetting.commission_percentage)
      const opportunityValue = parseFloat(opportunity.value)
      const commissionAmount = (opportunityValue * commissionPercentage) / 100

      // Create commission record
      await db.insert(commissions).values({
        org_id: orgId,
        user_id: opportunity.owner_user_id,
        opportunity_id: id,
        opportunity_value: opportunityValue.toString(),
        commission_percentage: commissionPercentage.toString(),
        commission_amount: commissionAmount.toString(),
        earned_date: new Date(),
        product_type_id: productTypeRecord.id, // Add product type to commission record
      })

      console.log(`[COMMISSION] ${opportunity.owner_user_id} earned $${commissionAmount} on ${productType} deal ${id}`)
    }
  }
}
```

## Task 7: Create Product Types API

Create endpoint: `artifacts/api-server/src/routes/productTypes.ts`

```typescript
import { db } from '@/db'
import { productTypes } from '@/db/schema'
import { eq, and } from 'drizzle-orm'

// GET product types for org
app.get('/api/product-types', async (req, res) => {
  const { orgId } = req.query
  const userId = req.user.id

  try {
    // Verify user is member of org
    const membership = await db.query.orgUsers.findFirst({
      where: and(
        eq(orgUsers.org_id, orgId),
        eq(orgUsers.user_id, userId),
        eq(orgUsers.status, 'active')
      ),
    })

    if (!membership) {
      return res.status(403).json({ error: 'Not member of organization' })
    }

    // Get all product types for org
    const types = await db.query.productTypes.findMany({
      where: and(
        eq(productTypes.org_id, orgId),
        eq(productTypes.is_active, true)
      ),
    })

    res.json(types)
  } catch (error) {
    console.error('Error fetching product types:', error)
    res.status(500).json({ error: 'Server error' })
  }
})

// CREATE new product type (admin only)
app.post('/api/product-types', async (req, res) => {
  const { orgId, name, description } = req.body
  const userId = req.user.id

  try {
    // Verify user is admin/owner
    const membership = await db.query.orgUsers.findFirst({
      where: and(
        eq(orgUsers.org_id, orgId),
        eq(orgUsers.user_id, userId)
      ),
    })

    if (!membership || (membership.role !== 'admin' && membership.role !== 'owner')) {
      return res.status(403).json({ error: 'Not authorized' })
    }

    // Create product type
    const newType = await db.insert(productTypes).values({
      org_id: orgId,
      name,
      description,
    })

    res.json({ success: true, type: newType })
  } catch (error) {
    console.error('Error creating product type:', error)
    res.status(500).json({ error: 'Server error' })
  }
})
```

## Task 9: Update Commission Settings API

Update the POST endpoint to handle new structure.

In `artifacts/api-server/src/routes/commissions.ts`:

**Replace the POST endpoint with:**

```typescript
app.post('/api/commissions/settings', async (req, res) => {
  const { orgId, commissions } = req.body // { "userId-productTypeId": percentage }
  const userId = req.user.id

  try {
    // Verify user is admin/owner
    const membership = await db.query.orgUsers.findFirst({
      where: and(
        eq(orgUsers.org_id, orgId),
        eq(orgUsers.user_id, userId)
      ),
    })

    if (!membership || (membership.role !== 'admin' && membership.role !== 'owner')) {
      return res.status(403).json({ error: 'Not authorized' })
    }

    // Update each employee's commission per product type
    for (const [key, percentage] of Object.entries(commissions)) {
      const [employeeId, productTypeId] = key.split('-')

      const existing = await db.query.employeeCommissions.findFirst({
        where: and(
          eq(employeeCommissions.org_id, orgId),
          eq(employeeCommissions.user_id, employeeId),
          eq(employeeCommissions.product_type_id, productTypeId)
        ),
      })

      if (existing) {
        // Update
        await db.update(employeeCommissions)
          .set({
            commission_percentage: percentage.toString(),
            updated_at: new Date(),
          })
          .where(eq(employeeCommissions.id, existing.id))
      } else {
        // Create
        await db.insert(employeeCommissions).values({
          org_id: orgId,
          user_id: employeeId,
          product_type_id: productTypeId,
          commission_percentage: percentage.toString(),
        })
      }
    }

    res.json({ success: true })
  } catch (error) {
    console.error('Error updating commissions:', error)
    res.status(500).json({ error: 'Server error' })
  }
})
```
