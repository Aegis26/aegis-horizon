# Aegis Horizon — The Most Ambitious Landing Page & Auth Refinement Brief

**Vision:** A landing page and authentication experience so visually stunning and technically sophisticated that users feel they've accessed a command center, not a form. Every pixel, every animation, every interaction tells them: *"This is built by serious people for serious work."*

**Standard:** Enterprise-grade visual design meets cutting-edge web animation. Think Vercel, Linear, or Stripe—but with the technical depth of a command-and-control interface. Premium micro-interactions, sophisticated animations, and architectural depth at every layer.

**Goal:** The moment a prospect lands on the page, they're captivated. By the time they complete signup, they're already invested in the experience and ready to see the product.

---

## 1. Visual Direction & Design Principles

### The Aesthetic: "Precision. Authority. Forward."

- **Color Palette**: Deep navy backgrounds with electric blue accents, white typography
- **Motion**: Subtle, purposeful, never arbitrary. Every animation serves function + delight
- **Depth**: Multiple visual layers create perceived depth—glassmorphism, shadows, layering
- **Typography**: Archivo for headings (geometric, commanding), Public Sans for body (clean, professional)
- **Imagery**: Geometric patterns, data visualization, tech-forward iconography
- **Responsiveness**: Premium experience on mobile—not a scaled-down desktop version

### Design Principles
1. **Contrast is King** — Electric blue on deep navy must be sharp and readable
2. **Whitespace is Intentional** — Every gap serves rhythm and clarity
3. **Motion is Purposeful** — No decoration without function
4. **Hierarchy is Invisible** — Users know what to focus on without thinking
5. **Performance is Assumed** — Animations run at 60fps, no jank
6. **Accessibility is Built-in** — Focus states, color contrast, keyboard nav, reduced-motion support

### Animated Text Reveal Component

```typescript
// components/AnimatedTextReveal.tsx
import { useEffect, useRef, useState } from 'react';

interface Props {
  text: string;
  staggerDelay?: number;
}

export function AnimatedTextReveal({ text, staggerDelay = 0.05 }: Props) {
  const [isVisible, setIsVisible] = useState(false);
  const ref = useRef(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
        }
      },
      { threshold: 0.1 }
    );

    if (ref.current) observer.observe(ref.current);
    return () => observer.disconnect();
  }, []);

  const words = text.split(' ');

  return (
    <span ref={ref} className="inline-block">
      {words.map((word, idx) => (
        <span
          key={idx}
          className="inline-block"
          style={{
            animation: isVisible ? `slideUpReveal 0.6s cubic-bezier(0.34, 1.56, 0.64, 1) forwards` : 'none',
            animationDelay: `${idx * staggerDelay}s`,
            opacity: isVisible ? 1 : 0,
            transform: isVisible ? 'translateY(0)' : 'translateY(20px)',
          }}
        >
          {word}&nbsp;
        </span>
      ))}
    </span>
  );
}
```

## 4. Floating Geometric Elements

### 3D-like Floating Cards

```typescript
// components/FloatingCard.tsx
import React, { useEffect, useState } from 'react';

interface FloatingCardProps {
  title: string;
  description: string;
  icon: React.ReactNode;
  delay?: number;
  position: 'top-left' | 'top-right' | 'bottom-left' | 'bottom-right';
}

export function FloatingCard({
  title,
  description,
  icon,
  delay = 0,
  position,
}: FloatingCardProps) {
  const [mousePosition, setMousePosition] = useState({ x: 0, y: 0 });
  const [isHovered, setIsHovered] = useState(false);

  useEffect(() => {
    const handleMouseMove = (e: MouseEvent) => {
      setMousePosition({ x: e.clientX, y: e.clientY });
    };

    window.addEventListener('mousemove', handleMouseMove);
    return () => window.removeEventListener('mousemove', handleMouseMove);
  }, []);

  const getPositionClass = () => {
    const positions = {
      'top-left': 'top-20 left-20',
      'top-right': 'top-20 right-20',
      'bottom-left': 'bottom-20 left-20',
      'bottom-right': 'bottom-20 right-20',
    };
    return positions[position];
  };

  return (
    <div
      className={`absolute w-72 ${getPositionClass()}`}
      style={{
        animation: `float 6s ease-in-out infinite`,
        animationDelay: `${delay}s`,
      }}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      <div
        className="
          relative
          p-6
          rounded-xl
          border border-primary border-opacity-20
          bg-gradient-to-br from-bg-dark to-bg-dark
          backdrop-blur-md
          shadow-lg
          transition-all duration-300
          cursor-pointer
          group
        "
        style={{
          transform: isHovered
            ? `translateY(-8px) scale(1.05) perspective(1200px)`
            : 'translateY(0) scale(1)',
          boxShadow: isHovered
            ? '0 20px 40px rgba(0, 180, 216, 0.2)'
            : '0 10px 30px rgba(0, 0, 0, 0.3)',
          background: isHovered
            ? 'linear-gradient(135deg, rgba(0, 180, 216, 0.1) 0%, rgba(0, 102, 204, 0.05) 100%)'
            : 'linear-gradient(135deg, rgba(10, 14, 39, 0.8) 0%, rgba(10, 14, 39, 0.6) 100%)',
        }}
      >
        {/* Gradient border on hover */}
        {isHovered && (
          <div
            className="absolute inset-0 rounded-xl opacity-0 group-hover:opacity-100 transition-opacity"
            style={{
              background:
                'linear-gradient(135deg, rgba(0, 180, 216, 0.3), rgba(0, 102, 204, 0.2))',
              pointerEvents: 'none',
            }}
          />
        )}

        {/* Content */}
        <div className="relative z-10">
          <div className="text-4xl mb-4 text-primary">{icon}</div>
          <h3 className="font-display text-label text-white mb-2">{title}</h3>
          <p className="text-small text-text-light text-opacity-70">
            {description}
          </p>
        </div>

        {/* Animated corner accent */}
        <div
          className="absolute top-0 right-0 w-20 h-20 border-t-2 border-r-2 border-primary rounded-tr-xl opacity-0 group-hover:opacity-100 transition-all"
          style={{
            animation: isHovered ? 'cornerGlow 0.6s ease' : 'none',
          }}
        />
      </div>
    </div>
  );
}
```

### Animated Input Component

```typescript
// components/AnimatedInput.tsx
import React, { useState, useRef } from 'react';

interface AnimatedInputProps {
  type: string;
  placeholder: string;
  icon?: string;
  showStrength?: boolean;
}

export function AnimatedInput({
  type,
  placeholder,
  icon,
  showStrength,
}: AnimatedInputProps) {
  const [isFocused, setIsFocused] = useState(false);
  const [value, setValue] = useState('');
  const inputRef = useRef<HTMLInputElement>(null);

  const getPasswordStrength = (val: string) => {
    if (val.length < 6) return { level: 0, text: 'Weak', color: 'error' };
    if (val.length < 10) return { level: 1, text: 'Fair', color: 'warning' };
    if (!/[!@#$%^&*]/.test(val)) return { level: 2, text: 'Good', color: 'info' };
    return { level: 3, text: 'Strong', color: 'success' };
  };

  const strength = showStrength ? getPasswordStrength(value) : null;

  return (
    <div className="relative">
      <div
        className="
          relative
          overflow-hidden
          rounded-lg
          border-2
          transition-all duration-200
          bg-bg-dark bg-opacity-50
        "
        style={{
          borderColor: isFocused ? 'rgba(0, 180, 216, 0.5)' : 'rgba(0, 180, 216, 0.2)',
          boxShadow: isFocused
            ? '0 0 20px rgba(0, 180, 216, 0.2)'
            : '0 0 0 rgba(0, 180, 216, 0)',
        }}
      >
        {/* Animated focus glow */}
        {isFocused && (
          <div
            className="absolute inset-0 pointer-events-none opacity-0 animate-pulse"
            style={{
              background:
                'linear-gradient(90deg, transparent, rgba(0, 180, 216, 0.1), transparent)',
            }}
          />
        )}

        {/* Input container */}
        <div className="flex items-center gap-3 px-4 py-3">
          {icon && (
            <div
              className="text-primary transition-all"
              style={{
                transform: isFocused ? 'scale(1.2)' : 'scale(1)',
                opacity: isFocused ? 1 : 0.6,
              }}
            >
              {getIconByName(icon)}
            </div>
          )}

          <input
            ref={inputRef}
            type={type}
            value={value}
            onChange={(e) => setValue(e.target.value)}
            onFocus={() => setIsFocused(true)}
            onBlur={() => setIsFocused(false)}
            placeholder={placeholder}
            className="
              flex-1
              bg-transparent
              outline-none
              text-text-light
              placeholder-text-light placeholder-opacity-40
              font-sans
              transition-all
            "
          />
        </div>
      </div>

      {/* Password strength indicator */}
      {showStrength && value && (
        <div className="mt-2">
          <div className="flex gap-1 mb-1">
            {[...Array(4)].map((_, i) => (
              <div
                key={i}
                className="flex-1 h-1 rounded-full bg-primary bg-opacity-20 overflow-hidden"
              >
                <div
                  className={`h-full transition-all ${
                    i <= strength!.level ? `bg-${strength!.color}` : ''
                  }`}
                  style={{
                    width: i <= strength!.level ? '100%' : '0%',
                  }}
                />
              </div>
            ))}
          </div>
          <p className={`text-tiny text-${strength!.color}`}>
            {strength!.text} password
          </p>
        </div>
      )}
    </div>
  );
}

function getIconByName(name: string) {
  const icons: Record<string, JSX.Element> = {
    user: <UserIcon size={20} />,
    mail: <MailIcon size={20} />,
    lock: <LockIcon size={20} />,
    building: <BuildingIcon size={20} />,

  return icons[name] || null;
```

### Animated Button Component

```typescript
// components/AnimatedButton.tsx
import React, { useState } from 'react';

interface AnimatedButtonProps {
  children: React.ReactNode;
  variant: 'primary' | 'secondary' | 'tertiary';
  size: 'sm' | 'md' | 'lg';
  isLoading?: boolean;
  onClick?: () => void | Promise<void>;
  className?: string;
}

export function AnimatedButton({
  children,
  variant,
  size,
  isLoading,
  onClick,
  className,
}: AnimatedButtonProps) {
  const [isHovered, setIsHovered] = useState(false);

  const variantStyles = {
    primary: 'bg-primary text-bg-dark hover:shadow-glow',
    secondary:
      'border border-primary border-opacity-30 text-primary hover:bg-primary hover:bg-opacity-10',
    tertiary: 'text-primary hover:bg-primary hover:bg-opacity-5',
  };

  const sizeStyles = {
    sm: 'px-3 py-1.5 text-label',
    md: 'px-5 py-2.5 text-label',
    lg: 'px-6 py-3 text-base',
  };

  return (
    <button
      onClick={onClick}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
      disabled={isLoading}
      className={`
        relative
        font-display font-weight-600
        rounded-lg
        transition-all duration-200
        overflow-hidden
        group
        ${variantStyles[variant]}
        ${sizeStyles[size]}
        ${className}
        disabled:opacity-50 disabled:cursor-not-allowed
      `}
    >
      {/* Shimmer effect on hover */}
      {isHovered && (
        <div
          className="absolute inset-0 opacity-0 group-hover:opacity-100 transition-opacity"
          style={{
            background: 'linear-gradient(90deg, transparent, rgba(255,255,255,0.2), transparent)',
            animation: 'shimmer 0.6s ease',
          }}
        />
      )}

      {/* Loading spinner */}
      {isLoading && (
        <div className="absolute inset-0 flex items-center justify-center">
          <div className="spinner" />
        </div>
      )}

      {/* Text with loading opacity */}
      <span className={`transition-opacity ${isLoading ? 'opacity-0' : 'opacity-100'}`}>
        {children}
      </span>
    </button>
  );
}
```
