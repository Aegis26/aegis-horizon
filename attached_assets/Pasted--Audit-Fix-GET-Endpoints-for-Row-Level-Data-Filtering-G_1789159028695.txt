# Audit & Fix GET Endpoints for Row-Level Data Filtering

**Goal:** Verify all GET endpoints filter data by user role and ownership (so employees see only their data, not owner's).

**Critical Issue:** Employee logged in sees owner's data instead of empty dashboard.

**Expected Behavior:**
- Admin/Owner: GET `/api/accounts` → Returns ALL accounts
- Employee: GET `/api/accounts` → Returns ONLY accounts they own
- Same for opportunities, contacts, leads, etc.

## Task 3: Fix Endpoints That Don't Filter

For any endpoint that's NOT filtering by user ownership, update it.

**Template for fixing GET /api/accounts:**

```typescript
// BEFORE (returns all data to any member)
app.get('/api/accounts', async (req, res) => {
  const { orgId } = req.query
  
  const accounts = await db.query.accounts.findMany({
    where: eq(accounts.org_id, orgId),
  })
  
  res.json(accounts)
})

// AFTER (filters by user role and ownership)
app.get('/api/accounts', async (req, res) => {
  const { orgId } = req.query
  const userId = req.user.id
  
  // Get user's role in org
  const membership = await db.query.orgUsers.findFirst({
    where: and(
      eq(orgUsers.org_id, orgId),
      eq(orgUsers.user_id, userId),
      eq(orgUsers.status, 'active')
    ),
  })
  
  if (!membership) {
    return res.status(403).json({ error: 'Not member of organization' })
  }
  
  let accounts
  
  // Admin and managers see all org accounts
  if (membership.role === 'admin' || membership.role === 'manager') {
    accounts = await db.query.accounts.findMany({
      where: eq(accounts.org_id, orgId),
    })
  }
  // Users see only their own accounts
  else if (membership.role === 'user') {
    accounts = await db.query.accounts.findMany({
      where: and(
        eq(accounts.org_id, orgId),
        or(
          eq(accounts.owner_user_id, userId),
          eq(accounts.created_by_user_id, userId)
        )
      ),
    })
  }
  // Viewers see only accounts they own (read-only)
  else if (membership.role === 'viewer') {
    accounts = await db.query.accounts.findMany({
      where: and(
        eq(accounts.org_id, orgId),
        eq(accounts.owner_user_id, userId)
      ),
    })
  }
  else {
    return res.status(403).json({ error: 'Invalid role' })
  }
  
  res.json(accounts)
})
```

**Apply same pattern to:**
- `GET /api/opportunities`
- `GET /api/contacts`
- `GET /api/leads`
- `GET /api/:id` (detail endpoints)
- Any other GET endpoints

## Task 4: Fix Detail Endpoints (GET by ID)

Detail endpoints need EXTRA CARE - must verify user can access that specific record:

```typescript
// BEFORE (returns any account by ID, no permission check)
app.get('/api/accounts/:id', async (req, res) => {
  const { id } = req.params
  
  const account = await db.query.accounts.findFirst({
    where: eq(accounts.id, id),
  })
  
  res.json(account)
})

// AFTER (checks user can access this specific account)
app.get('/api/accounts/:id', async (req, res) => {
  const { id } = req.params
  const { orgId } = req.query
  const userId = req.user.id
  
  // Get user's role
  const membership = await db.query.orgUsers.findFirst({
    where: and(
      eq(orgUsers.org_id, orgId),
      eq(orgUsers.user_id, userId),
      eq(orgUsers.status, 'active')
    ),
  })
  
  if (!membership) {
    return res.status(403).json({ error: 'Not member' })
  }
  
  // Get the account
  const account = await db.query.accounts.findFirst({
    where: and(
      eq(accounts.id, id),
      eq(accounts.org_id, orgId)
    ),
  })
  
  if (!account) {
    return res.status(404).json({ error: 'Account not found' })
  }
  
  // Check access permission
  if (membership.role === 'admin' || membership.role === 'manager') {
    // Can view any org account
    res.json(account)
  } else if (membership.role === 'user') {
    // Can only view own accounts
    if (account.owner_user_id !== userId && account.created_by_user_id !== userId) {
      return res.status(403).json({ error: 'Access denied' })
    }
    res.json(account)
  } else if (membership.role === 'viewer') {
    // Can only view owned accounts
    if (account.owner_user_id !== userId) {
      return res.status(403).json({ error: 'Access denied' })
    }
    res.json(account)
  }
})
```

## Task 5: Helper Function (Optional but Recommended)

Create a reusable filter helper in `artifacts/api-server/src/lib/dataAccessControl.ts`:

```typescript
import { db } from '@/db'
import { orgUsers } from '@/db/schema'
import { and, eq, or } from 'drizzle-orm'

export async function getUserMembership(orgId: string, userId: string) {
  return await db.query.orgUsers.findFirst({
    where: and(
      eq(orgUsers.org_id, orgId),
      eq(orgUsers.user_id, userId),
      eq(orgUsers.status, 'active')
    ),
  })
}

export function buildVisibilityFilter(
  table: any,
  userRole: string,
  userId: string,
  orgId: string
) {
  if (userRole === 'admin' || userRole === 'manager') {
    return eq(table.org_id, orgId)
  } else if (userRole === 'user') {
    return and(
      eq(table.org_id, orgId),
      or(
        eq(table.owner_user_id, userId),
        eq(table.created_by_user_id, userId)
      )
    )
  } else if (userRole === 'viewer') {
    return and(
      eq(table.org_id, orgId),
      eq(table.owner_user_id, userId)
    )
  }
  return null
}
```

Then use in endpoints:

```typescript
import { getUserMembership, buildVisibilityFilter } from '@/lib/dataAccessControl'

app.get('/api/accounts', async (req, res) => {
  const { orgId } = req.query
  const userId = req.user.id
  
  const membership = await getUserMembership(orgId, userId)
  if (!membership) return res.status(403).json({ error: 'Not member' })
  
  const filter = buildVisibilityFilter(accounts, membership.role, userId, orgId)
  const results = await db.query.accounts.findMany({ where: filter })
  
  res.json(results)
})
```
