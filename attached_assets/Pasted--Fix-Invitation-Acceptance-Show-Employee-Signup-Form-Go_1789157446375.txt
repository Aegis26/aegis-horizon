4. Pass it to the signup form

```typescript
const invitation = await db.query.orgUsers.findFirst({
```

## Task 3: Create Signup Form Component

Create or update the invitation page to show a signup form instead of account confirmation.

**File:** `artifacts/crm/src/pages/invite/[token].tsx`

```typescript
import { useState, useEffect } from 'react'
import { useRouter } from 'next/router'
import { useSignUp } from '@clerk/nextjs'

export default function InvitationSignupPage() {
  const router = useRouter()
  const { token } = router.query
  const { signUp, setActive } = useSignUp()
  
  const [invitedEmail, setInvitedEmail] = useState('')
  const [formData, setFormData] = useState({
    firstName: '',
    lastName: '',
    password: '',
    confirmPassword: '',
  })
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState('')
  const [status, setStatus] = useState('loading') // loading | form | success | error

  // Step 1: Fetch invited email from token
  useEffect(() => {
    if (!token) return

    const fetchInvitedEmail = async () => {
      try {
        const response = await fetch(`/api/invitations/email?token=${token}`)
        if (!response.ok) {
          setError('Invalid or expired invitation')
          setStatus('error')
          return
        }
        const data = await response.json()
        setInvitedEmail(data.email)
        setStatus('form')
      } catch (err) {
        setError('Failed to load invitation')
        setStatus('error')
      }
    }

    fetchInvitedEmail()
  }, [token])

  // Step 2: Handle signup
  const handleSignup = async (e) => {
    e.preventDefault()
    setLoading(true)
    setError('')

    try {
      // Validate passwords match
      if (formData.password !== formData.confirmPassword) {
        setError('Passwords do not match')
        setLoading(false)
        return
      }

      // Validate password strength
      if (formData.password.length < 8) {
        setError('Password must be at least 8 characters')
        setLoading(false)
        return
      }

      // Create Clerk account
      const result = await signUp.create({
        emailAddress: invitedEmail,
        password: formData.password,
        firstName: formData.firstName,
        lastName: formData.lastName,
        unsafeMetadata: {
          invitationToken: token,
        },
      })

      // Complete email verification (auto-verify for invitations)
      await setActive({ session: result.createdSessionId })

      // Step 3: Accept the invitation on backend
      setStatus('success')
      
      const acceptResponse = await fetch('/api/invitations/accept', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          token,
          userEmail: invitedEmail,
        }),
      })

      if (!acceptResponse.ok) {
        const data = await acceptResponse.json()
        throw new Error(data.error)
      }

      const invitationData = await acceptResponse.json()

      // Step 4: Redirect to their dashboard
      setTimeout(() => {
        router.push(`/dashboard?orgId=${invitationData.orgId}`)
      }, 1500)
    } catch (err) {
      console.error('Signup error:', err)
      setError(err.message || 'Failed to create account')
      setStatus('form')
      setLoading(false)
    }
  }

  // Handle input change
  const handleChange = (e) => {
    const { name, value } = e.target
    setFormData(prev => ({
      ...prev,
      [name]: value,
    }))
  }

  // Loading state
  if (status === 'loading') {
    return (
      <div className="min-h-screen bg-gradient-to-br from-slate-900 to-slate-800 flex items-center justify-center">
        <div className="text-center">
          <p className="text-white text-lg">Loading invitation...</p>
        </div>
      </div>
    )
  }

  // Error state
  if (status === 'error') {
    return (
      <div className="min-h-screen bg-gradient-to-br from-slate-900 to-slate-800 flex items-center justify-center">
        <div className="bg-slate-800 border border-red-500 rounded-lg p-8 max-w-md text-center">
          <h1 className="text-2xl font-bold text-red-400 mb-4">Invitation Error</h1>
          <p className="text-gray-300 mb-6">{error}</p>
          <button 
            onClick={() => router.push('/')}
            className="bg-blue-500 hover:bg-blue-600 text-white px-6 py-2 rounded"
          >
            Go Home
          </button>
        </div>
      </div>
    )
  }

  // Success state
  if (status === 'success') {
    return (
      <div className="min-h-screen bg-gradient-to-br from-slate-900 to-slate-800 flex items-center justify-center">
        <div className="bg-slate-800 border border-green-500 rounded-lg p-8 max-w-md text-center">
          <h1 className="text-2xl font-bold text-green-400 mb-4">✓ Account Created!</h1>
          <p className="text-gray-300 mb-4">Welcome to Aegis Horizon</p>
          <p className="text-gray-400 text-sm">Redirecting to your dashboard...</p>
        </div>
      </div>
    )
  }

  // Signup form
  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-900 to-slate-800 flex items-center justify-center p-4">
      <div className="bg-slate-800 border border-slate-700 rounded-lg p-8 max-w-md w-full">
        <div className="mb-6">
          <h1 className="text-2xl font-bold text-white mb-2">Create Your Account</h1>
          <p className="text-gray-400 text-sm">Join Aegis Horizon</p>
        </div>

        <form onSubmit={handleSignup} className="space-y-4">
          {/* Email (read-only) */}
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-2">Email</label>
            <input
              type="email"
              value={invitedEmail}
              disabled
              className="w-full px-4 py-2 bg-slate-700 text-gray-400 rounded border border-slate-600 cursor-not-allowed"
            />
          </div>

          {/* First Name */}
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-2">First Name</label>
            <input
              type="text"
              name="firstName"
              value={formData.firstName}
              onChange={handleChange}
              required
              className="w-full px-4 py-2 bg-slate-700 text-white rounded border border-slate-600 focus:border-blue-500 focus:outline-none"
              placeholder="John"
            />
          </div>

          {/* Last Name */}
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-2">Last Name</label>
            <input
              type="text"
              name="lastName"
              value={formData.lastName}
              onChange={handleChange}
              required
              className="w-full px-4 py-2 bg-slate-700 text-white rounded border border-slate-600 focus:border-blue-500 focus:outline-none"
              placeholder="Doe"
            />
          </div>

          {/* Password */}
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-2">Password</label>
            <input
              type="password"
              name="password"
              value={formData.password}
              onChange={handleChange}
              required
              className="w-full px-4 py-2 bg-slate-700 text-white rounded border border-slate-600 focus:border-blue-500 focus:outline-none"
              placeholder="••••••••"
            />
            <p className="text-xs text-gray-500 mt-1">At least 8 characters</p>
          </div>

          {/* Confirm Password */}
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-2">Confirm Password</label>
            <input
              type="password"
              name="confirmPassword"
              value={formData.confirmPassword}
              onChange={handleChange}
              required
              className="w-full px-4 py-2 bg-slate-700 text-white rounded border border-slate-600 focus:border-blue-500 focus:outline-none"
              placeholder="••••••••"
            />
          </div>

          {/* Error message */}
          {error && (
            <div className="p-3 bg-red-500/10 border border-red-500 rounded text-red-400 text-sm">
              {error}
            </div>
          )}

          {/* Submit button */}
          <button
            type="submit"
            disabled={loading}
            className="w-full bg-blue-500 hover:bg-blue-600 disabled:bg-blue-400 text-white font-medium py-2 rounded transition"
          >
            {loading ? 'Creating Account...' : 'Create Account & Accept Invitation'}
          </button>
        </form>

        <p className="text-center text-gray-500 text-xs mt-6">
          You'll be logged in and redirected to your dashboard
        </p>
      </div>
    </div>
  )
}
```

## Task 4: Create Backend Endpoint to Get Invited Email

Create endpoint: `artifacts/api-server/src/routes/invitations.ts` (or add to existing):

```typescript
import { db } from '@/db'
import { orgUsers } from '@/db/schema'
import { eq } from 'drizzle-orm'
import crypto from 'crypto'

app.get('/api/invitations/email', async (req, res) => {
  const { token } = req.query

  if (!token || typeof token !== 'string') {
    return res.status(400).json({ error: 'Token required' })
  }

  try {
    // Hash the token
    const tokenHash = crypto.createHash('sha256').update(token).digest('hex')

    // Find the invitation
    const invitation = await db.query.orgUsers.findFirst({
      where: and(
        eq(orgUsers.invitation_token_hash, tokenHash),
        eq(orgUsers.status, 'pending')
      ),
    })

    if (!invitation) {
      return res.status(404).json({ error: 'Invitation not found or expired' })
    }

    // Check expiry
    if (invitation.invitation_expires_at && invitation.invitation_expires_at < new Date()) {
      return res.status(410).json({ error: 'Invitation expired' })
    }

    // Return the invited email
    res.json({ email: invitation.email })
  } catch (error) {
    console.error('Error fetching invitation email:', error)
    res.status(500).json({ error: 'Server error' })
  }
})
```

```typescript
app.post('/api/invitations/accept', async (req, res) => {
  const { token, userEmail } = req.body

  if (!token || !userEmail) {
    return res.status(400).json({ error: 'Token and email required' })
  }

  try {
    // Hash the token
    const tokenHash = crypto.createHash('sha256').update(token).digest('hex')

    // Find invitation
    const invitation = await db.query.orgUsers.findFirst({
      where: and(
        eq(orgUsers.email, userEmail),
        eq(orgUsers.status, 'pending'),
        eq(orgUsers.invitation_token_hash, tokenHash)
      ),
    })

    if (!invitation) {
      return res.status(404).json({ error: 'Invitation not found' })
    }

    // Check expiry
    if (invitation.invitation_expires_at < new Date()) {
      return res.status(410).json({ error: 'Invitation expired' })
    }

    // Accept invitation (already have user from Clerk)
    const user = await clerk.users.getUser(invitation.user_id) // Get from Clerk

    await db.update(orgUsers)
      .set({
        status: 'active',
        invitation_accepted_at: new Date(),
        invitation_token_hash: null, // Single-use
        invitation_expires_at: null,
      })
      .where(eq(orgUsers.id, invitation.id))

    // Log acceptance
    await AuditService.log(
      invitation.org_id,
      invitation.user_id,
      'accept_invitation',
      'org_user',
      invitation.id
    )

    // Get org info
    const org = await db.query.organizations.findFirst({
      where: eq(organizations.id, invitation.org_id),
    })

    res.json({
      success: true,
      orgId: invitation.org_id,
      orgName: org.name,
    })
  } catch (error) {
    console.error('Invitation acceptance error:', error)
    res.status(500).json({ error: 'Server error' })
  }
})
```
